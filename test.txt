            localStorage.setItem('dicomStudyMap', JSON.stringify(dicomStudyMap));

        // Modal control functions
        function openModal(modalId) { 
    const modal = document.getElementById(modalId);
    if (modal) {
        modal.classList.add('visible');
        // Prevent background scroll when nested modal is open
        document.body.style.overflow = 'hidden';
    } else {
        console.error(`Modal avec ID ${modalId} non trouvé`);
    }
}

function closeModal(modalId) { 
    const modal = document.getElementById(modalId);
    if (modal) {
        modal.classList.remove('visible');
        // Restore scroll if no other modal is open
        const activeModals = document.querySelectorAll('.modal.visible');
        if (activeModals.length === 0) {
            document.body.style.overflow = 'auto';
        }
    } else {
        console.error(`Modal avec ID ${modalId} non trouvé`);
    }
}

        function closeUploadDicomModal() { closeModal('upload-dicom-modal'); }

                if (!file) { alert('Veuillez sélectionner un fichier DICOM.'); return; }
                const formData = new FormData(); formData.append('file', file);
                try {
                    const response = await fetch(`${ORTHANC_PROXY_URL}?path=instances`, { method: 'POST', body: formData });
                    const text = await response.text();
                    const result = JSON.parse(text);

            document.getElementById('upload-dicom-patient-form')?.addEventListener('submit', async (e) => {

                const patientId = document.getElementById('upload-dicom-patient-id').value;
                const consultationId = e.target.dataset.consultationId;
                const file = document.getElementById('upload-dicom-patient-file').files[0];
                const timestamp = new Date().toISOString().replace(/[-:T.]/g, '').slice(0, 14);
                const instanceId = `PATIENT-${patientId}-${timestamp}`;
                if (!file || !patientId) { alert('Fichier ou patient manquant.'); return; }
                const formData = new FormData();
                formData.append('file', file);
                formData.append('patient_id', patientId);
                formData.append('instance_id', instanceId);
                if (consultationId) {
                    formData.append('consultation_id', consultationId);
                }
                try {
                    const response = await fetch(`${ORTHANC_PROXY_URL}?path=instances`, { 
                        method: 'POST', 
                        body: formData,
                        headers: { 'Authorization': `Bearer ${token}` }
                    });
                    const text = await response.text();
                    const result = JSON.parse(text);
                    if (!response.ok) throw new Error(result.error || 'Erreur d\'upload');

                    saveDicomStudyMap();

            document.getElementById('medical-record-patient-id').value = patientId;

        // Open consultations modal
        async function openConsultationsModal(patientId) {
    console.log(`Ouverture des consultations pour patientId: ${patientId}`);
    const notesContent = document.getElementById('consultations-content');
    const dicomContent = document.getElementById('dicom-content');
    if (!notesContent || !dicomContent) {
        console.error('Élément de contenu des consultations ou DICOM non trouvé');
        alert('Erreur: Élément de contenu non trouvé.');
        return;
    }

    if (!token || !token.trim()) {
        console.error('Token manquant ou vide dans openConsultationsModal');
        alert('Session expirée. Veuillez vous reconnecter.');
        logout();
        return;
    }

    console.log('Token avant requête DICOM:', token || 'Token est vide/null');

    document.getElementById('consultations-patient-id').value = patientId;
    notesContent.innerHTML = '<p class="text-gray-400">Chargement...</p>';
    dicomContent.innerHTML = '<p class="text-gray-400">Chargement...</p>';
    openModal('consultations-modal');

    const headers = {
        'Authorization': `Bearer ${token}`,
        'Content-Type': 'application/json',
        'Accept': 'application/json'
    };

    try {
        const consultationsResponse = await fetch(`http://localhost/projet-medical/api/consultations.php?patientId=${patientId}`, {
            method: 'GET',
            headers: headers
        });

        if (!consultationsResponse.ok) {
            const errorText = await consultationsResponse.text();
            console.error(`Erreur HTTP ${consultationsResponse.status} (consultations): ${errorText}`);
            if (consultationsResponse.status === 401) {
                alert('Session expirée. Veuillez vous reconnecter.');
                logout();
                return;
            }
            throw new Error(`Erreur HTTP ${consultationsResponse.status}: ${errorText}`);
        }

        const consultations = await consultationsResponse.json();
        console.log('Consultations récupérées:', consultations);

        if (consultations.length === 0) {
            notesContent.innerHTML = '<p class="text-gray-400">Aucune consultation trouvée.</p>';
        } else {
            notesContent.innerHTML = consultations.map(c => `
                <div class="timeline-item cursor-pointer" onclick="openConsultationDetails(${c.id}, '${c.date_consultation}', \`${c.diagnostic}\`)">
                    <div class="timeline-icon"><i class="fas fa-stethoscope"></i></div>
                    <div class="timeline-content">
                        <h4>Consultation du ${new Date(c.date_consultation).toLocaleString('fr-FR')}</h4>
                    </div>
                </div>
            `).join('');
        }

        const dicomResponse = await fetch(`http://localhost/projet-medical/api/dicom-instances.php?patientId=${patientId}`, {
            method: 'GET',
            headers: headers
        });

        if (!dicomResponse.ok) {
            const errorText = await dicomResponse.text();
            console.error(`Erreur HTTP ${dicomResponse.status} (DICOM): ${errorText}`);
            if (dicomResponse.status === 401) {
                alert('Session expirée ou accès non autorisé. Veuillez vous reconnecter.');
                logout();
                return;
            }
            if (dicomResponse.status === 403) {
                dicomContent.innerHTML = '<p class="text-red-400">Accès refusé : Vous n\'êtes pas autorisé à voir les fichiers DICOM de ce patient.</p>';
                return;
            }
            throw new Error(`Erreur HTTP ${dicomResponse.status}: ${errorText}`);
        }

        const dicomData = await dicomResponse.json();
        const dicomInstances = dicomData.dicom_instances || [];
        console.log('DICOM instances récupérées:', dicomInstances);

        if (dicomInstances.length === 0) {
            dicomContent.innerHTML = '<p class="text-gray-400">Aucun fichier DICOM trouvé.</p>';
        } else {
            dicomContent.innerHTML = dicomInstances.map(d => `
                <div class="dicom-card">
                    <i class="fas fa-file-medical mr-3"></i>
                    <div>
                        <p class="text-white">${d.description || 'N/A'}</p>
                        <p class="text-gray-400 text-sm">Consultation ID: ${d.consultation_id || 'N/A'}</p>
                        <p class="text-gray-400 text-sm">Upload: ${new Date(d.upload_date).toLocaleString('fr-FR')}</p>
                    </div>
                </div>
            `).join('');
        }
    } catch (error) {
        console.error('Erreur lors de la récupération des consultations ou DICOM:', error);
        notesContent.innerHTML = `<p class="text-red-400">Erreur: ${error.message}</p>`;
        dicomContent.innerHTML = `<p class="text-red-400">Erreur: ${error.message}</p>`;
        alert('Erreur lors de la récupération des données: ' + error.message);
    }
}
